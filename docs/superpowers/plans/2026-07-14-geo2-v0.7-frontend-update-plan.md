```tsx
// frontend/src/components/flow/TruncationBadge.tsx
import type { TruncationStrategy } from '@/types/v0.7';

const LABELS: Record<TruncationStrategy, string> = {
  noop: '压缩无需',
  truncate: '截断',
  drop: '丢弃',
  summarize: '摘要',
};

const CLASSES: Record<TruncationStrategy, string> = {
  noop: 'bg-gray-100 text-gray-600',
  truncate: 'bg-primary-tint text-primary',
  drop: 'bg-warning/20 text-warning',
  summarize: 'bg-accent/30 text-accent',
};

export function TruncationBadge({ strategy, savedTokens }: { strategy: TruncationStrategy; savedTokens: number }) {
  if (strategy === 'noop') {
    return <span className={`rounded-md px-2 py-1 text-xs ${CLASSES.noop}`}>{LABELS.noop}</span>;
  }
  return (
    <span className={`rounded-md px-2 py-1 text-xs ${CLASSES[strategy]}`}>
      {LABELS[strategy]} · 节省 {(savedTokens / 1000).toFixed(1)}k tokens
    </span>
  );
}
```

```ts
// frontend/src/api/client.ts (修改部分)
export class ApiError extends Error {
  constructor(public status: number, message: string, public code?: string) {
    super(message);
    this.name = 'ApiError';
  }
}

async function request<T>(path: string, init: RequestOptions = {}): Promise<T> {
  const { signal, ...rest } = init;
  const resp = await fetch(`${BASE}${path}`, {
    headers: authHeaders(rest.headers),
    signal,
    ...rest,
  });
  if (!resp.ok) {
    const body = await resp.text();
    let code: string | undefined;
    try { code = JSON.parse(body).error_code; } catch {}
    throw new ApiError(resp.status, body || resp.statusText, code);
  }
  if (resp.status === 204) return undefined as T;
  return resp.json();
}
```

```tsx
// frontend/src/hooks/useAgentStream.ts
import { useState, useRef, useCallback } from 'react';
import type { AgentEvent } from '@/types/v0.4';
import { sendAgentMessageStream } from '@/api/agent';
import { useLangGraphMode } from './useLangGraphMode';
import type { HandoffLog, TruncationDecision, TruncationStrategy } from '@/types/v0.7';

export interface AgentStreamState {
  messages: Array<{ id: string; role: 'user' | 'assistant' | 'tool'; content: string }>;
  tools: Array<{ name: string; args: unknown; result?: unknown }>;
  handoffs: HandoffLog[];
  truncations: TruncationDecision[];
  memoryCount: number;
  status: 'idle' | 'streaming' | 'error';
  error?: string;
}

export function useAgentStream(sessionId: string) {
  const [state, setState] = useState<AgentStreamState>({
    messages: [], tools: [], handoffs: [], truncations: [], memoryCount: 0, status: 'idle',
  });
  const ctrlRef = useRef<AbortController | null>(null);
  const { enabled: langGraphMode } = useLangGraphMode();

  const send = useCallback(async (content: string) => {
    ctrlRef.current = new AbortController();
    setState((s) => ({ ...s, status: 'streaming', messages: [...s.messages, { id: crypto.randomUUID(), role: 'user', content }] }));
    try {
      for await (const evt of sendAgentMessageStream(sessionId, content)) {
        if (langGraphMode) {
          // LangGraph schema 分支(spec §6.2)
          setState((s) => applyLangGraphEvent(s, evt));
        } else {
          // react_loop 兼容分支(沿用 v0.6 P0 解析)
          setState((s) => applyReactLoopEvent(s, evt));
        }
      }
      setState((s) => ({ ...s, status: 'idle' }));
    } catch (e: any) {
      setState((s) => ({ ...s, status: 'error', error: e.message }));
    }
  }, [sessionId, langGraphMode]);

  const abort = useCallback(() => ctrlRef.current?.abort(), []);

  return { ...state, send, abort };
}

function applyLangGraphEvent(s: AgentStreamState, evt: AgentEvent): AgentStreamState {
  switch (evt.event) {
    case 'handoff':
      return { ...s, handoffs: [...s.handoffs, {
        id: crypto.randomUUID(), sessionId: '',
        fromAgent: evt.from_agent, toAgent: evt.to_agent,
        taskSummary: evt.task_summary, reason: evt.reason, timestamp: evt.timestamp,
      }]};
    case 'truncation_decision':
      return { ...s, truncations: [...s.truncations, {
        strategy: evt.strategy as TruncationStrategy,
        originalTokens: 0, finalTokens: 0, savedTokens: evt.saved_tokens,
      }]};
    case 'memory_injected':
      return { ...s, memoryCount: evt.count };
    case 'llm_token':
      // 累积到最后一个 assistant message(简化)
      return s;
    case 'tool_call':
    case 'tool_result':
      return s; // ChatMessage 流处理
    default:
      return s;
  }
}
```

```ts
// frontend/src/api/agent.ts(新增)
export async function* replayAgentMessage(
  sessionId: string, msgId: string, signal?: AbortSignal,
): AsyncGenerator<AgentEvent> {
  const resp = await fetch(`${BASE}/agent/sessions/${sessionId}/replay/${msgId}`, {
    method: 'POST',
    headers: authHeaders(),
    body: JSON.stringify({}),
    signal,
  });
  if (!resp.ok) {
    const text = await resp.text();
    throw new ApiError(resp.status, text || resp.statusText);
  }
  if (!resp.body) return;
  const reader = resp.body.getReader();
  const decoder = new TextDecoder();
  let buffer = '';
  while (true) {
    const { value, done } = await reader.read();
    if (done) break;
    buffer += decoder.decode(value, { stream: true });
    const { events, remainder } = parseSSE(buffer);
    buffer = remainder;
    for (const evt of events) yield evt;
  }
}
```
